🔐 6. SYSTÈME DE PERMISSIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━

🔐 Fichiers clés
  src/lib/auth.tsx                              [Context Auth client]
  src/lib/api-auth.ts                           [Auth check serveur/API]

📊 Types de rôles
  UserRole: "coach" | "player" | "parent"
  TeamMemberRole: "owner" | "coach" | "player" | "parent"

🗄️ Tables BD
  profiles {
    id, role, first_name, last_name, position, shirt_number, vma, vmi
  }
  clubs {
    id, name, logo_url, created_by
  }
  teams {
    id, club_id, name, invite_code, color_primary, color_secondary
  }
  team_members {
    id, team_id, user_id, role,                 ← CLEF
    UNIQUE(team_id, user_id)
  }

🔧 Fonctions API
  getAuthUser(req)
    → Extrait Bearer token
    → Retourne User | null

  getTeamRole(userId, teamId)
    → Retourne "owner" | "coach" | "player" | "parent" | null

  isTeamCoach(userId, teamId)
    → true si role === "owner" || role === "coach"

  isTeamMember(userId, teamId)
    → true si exists in team_members

  getUserTeamIds(userId)
    → string[] de tous les teams

🔐 RLS (Row Level Security)
  Chaque table filtre: 
    team_id IN (SELECT team_id FROM team_members WHERE user_id = auth.uid())
